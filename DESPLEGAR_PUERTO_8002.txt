═══════════════════════════════════════════════════════════════

📋 EJECUTA ESTOS COMANDOS:

# 1. Actualizar servicio systemd al puerto 8002
sudo cp /tmp/edumind-robotics-port8002.service /etc/systemd/system/edumind-robotics.service

# 2. Actualizar configuración de Nginx
sudo cp /tmp/edumind-robotics-port8002.nginx /etc/nginx/sites-available/edumind-robotics

# 3. Verificar configuración de Nginx
sudo nginx -t

# 4. Recargar systemd
sudo systemctl daemon-reload

# 5. Reiniciar backend
sudo systemctl restart edumind-robotics

# 6. Recargar Nginx
sudo systemctl reload nginx

# 7. Esperar y verificar backend
sleep 3 && sudo systemctl status edumind-robotics --no-pager

# 8. Verificar que escucha en puerto 8002
sudo lsof -i :8002
