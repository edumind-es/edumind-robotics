# 1. Copiar configuración de Nginx
sudo cp /tmp/edumind-robotics.nginx /etc/nginx/sites-available/edumind-robotics

# 2. Activar sitio en Nginx
sudo ln -sf /etc/nginx/sites-available/edumind-robotics /etc/nginx/sites-enabled/edumind-robotics

# 3. Verificar configuración de Nginx
sudo nginx -t

# 4. Copiar servicio systemd
sudo cp /tmp/edumind-robotics.service /etc/systemd/system/edumind-robotics.service

# 5. Ajustar permisos del frontend
sudo chown -R www-data:www-data /var/www/edumind_robotics/frontend/dist/

# 6. Recargar systemd
sudo systemctl daemon-reload

# 7. Habilitar servicio para inicio automático
sudo systemctl enable edumind-robotics

# 8. Iniciar servicio backend
sudo systemctl start edumind-robotics

# 9. Recargar Nginx
sudo systemctl reload nginx

# 10. Verificar que todo funciona
sudo systemctl status edumind-robotics --no-pager
sudo systemctl status nginx --no-pager

COMANDOS ÚTILES DESPUÉS DEL DESPLIEGUE:

# Ver logs del backend en tiempo real
sudo journalctl -u edumind-robotics -f

# Reiniciar backend
sudo systemctl restart edumind-robotics

# Ver logs de Nginx
sudo tail -f /var/log/nginx/access.log
sudo tail -f /var/log/nginx/error.log

# Parar servicios
sudo systemctl stop edumind-robotics

# Ver estado de servicios
sudo systemctl status edumind-robotics
sudo systemctl status nginx
